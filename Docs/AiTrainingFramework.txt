class Training
{

	SetNextDifficultyLevel();
	bool CheckLevelGoal();
	bool CheckTrainingGoal();
	lambda GenerateAwardFunction();
}


class AdditionTraining : Training
{
    int numDigits = 1;
    
    void SetNextDifficultyLevel() 
    { 
        numDigits++;
	totalTrainedLoops = 0;
	TargetLearningLoops = numDigits * ExpectedTrainingCycleFactor;

        
        // 关键：难度变化时，训练目标也变化
        if (numDigits == 4)
        {
            // 引入新工具，训练目标扩展
            availableTools.Add("vertical_addition");
            evaluationCriteria.Add("tool_usage_appropriateness");
	 
        }	
    }
    // when training first starts, the award might encourage confidence to stay around 0.5 to avoid surprise penalty, that is correct, 
    // as any beginner should start cautions to a new field, but not completely cold feet
    // As trainings goes on, it has to improve, or it is an indication of training design complete failure
    // Must force AI improve confidence. But we don't need to worry about low capability as it is still heavily punished via surprise factor
    Func<float> GenerateAwardFunction()
    {
	    float progressRatio = (float)totalTrainedLoops / TargetLearningLoops;
	    float confidencePressure = (1.0 - confidence) * progressRatio * 0.5f;  // 0→0.5 线性增压
        // 动态奖励函数：根据当前难度调整权重
        if (numDigits <= 3)
        {
            // 简单阶段：只看结果和信心
            return () => 
            {
                float actual = IsAnswerCorrect ? 1f : 0f;
                float surprise = Math.Abs(confidence - actual);
                return -surprise * 2 - confidencePressure;  // 答错时重罚盲目自信
            };
        }
        else
        {
            // 复杂阶段：看过程、工具使用、步骤正确性
            return () =>
            {
                float reward = 0f;
                
                // 1. 结果和信心（权重降低）
                float actual = IsAnswerCorrect ? 1f : 0f;
                float surprise = Math.Abs(confidence - actual);
                reward += -surprise * 0.5f;
                
                // 2. 工具选择是否恰当
                bool shouldUseTool = numDigits >= 4;
                bool didUseTool = agent.UsedTool == "vertical_addition";
                
                if (shouldUseTool && didUseTool)
                    reward += 1.0f;      // 该用且用了：大奖
                else if (shouldUseTool && !didUseTool)
                    reward -= 0.8f;      // 该用却硬撑：惩罚
                else if (!shouldUseTool && didUseTool)
                    reward -= 0.3f;      // 不该用却用：轻微惩罚（资源浪费）
                
                // 3. 步骤正确性（如果用了工具）
                if (didUseTool)
                {
                    float stepAccuracy = EvaluateVerticalSteps(agent.Working);
                    reward += stepAccuracy * 0.5f;
                }
                
                // 4. 资源效率
                float costRatio = agent.ActualCost / Math.Max(agent.EstimatedCost, 1);
                if (costRatio > 2f)
                    reward -= 0.3f;      // 严重超预算
                    
                return reward - confidencePressure;
            };
        }
    }
}
